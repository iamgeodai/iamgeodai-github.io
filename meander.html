<script>

  const PARAMS = {

    initialFlow: 3,

    slope: 0.4,

    bankResistance: 1.0,

    timeStepYears: 1,

    erosionBaseRate: 0.08,

    depositionBaseRate: 0.05,

    cutoffDistance: 18,

    targetSinuosity: 1.8,

    maxYears: 200

  };


  const canvas = document.getElementById("riverCanvas");

  const ctx = canvas.getContext("2d");


  let riverPoints = [];

  let oxbowLakes = [];

  let selectedIndex = -1;

  let running = false;

  let timerId = null;

  let year = 0;

  let gameOver = false;

  let winStatus = null;

  let town = { x: 450, y: 220 };


  let globalFlow = PARAMS.initialFlow;

  let globalSlope = PARAMS.slope;

  let globalBankResistance = PARAMS.bankResistance;


  const yearDisplay = document.getElementById("yearDisplay");

  const sinuosityDisplay = document.getElementById("sinuosityDisplay");

  const targetSinuosityDisplay = document.getElementById("targetSinuosityDisplay");

  const townDistanceDisplay = document.getElementById("townDistanceDisplay");

  const messageBox = document.getElementById("messageBox");


  const flowSlider = document.getElementById("flowSlider");

  const slopeSlider = document.getElementById("slopeSlider");

  const bankResSlider = document.getElementById("bankResSlider");

  const localErosionSlider = document.getElementById("localErosionSlider");


  const flowValue = document.getElementById("flowValue");

  const slopeValue = document.getElementById("slopeValue");

  const bankResValue = document.getElementById("bankResValue");

  const localErosionValue = document.getElementById("localErosionValue");


  const startBtn = document.getElementById("startBtn");

  const pauseBtn = document.getElementById("pauseBtn");

  const resetBtn = document.getElementById("resetBtn");


  function initSliders() {

    flowSlider.value = globalFlow;

    slopeSlider.value = globalSlope;

    bankResSlider.value = globalBankResistance;

    flowValue.textContent = globalFlow.toFixed(1);

    slopeValue.textContent = globalSlope.toFixed(1);

    bankResValue.textContent = globalBankResistance.toFixed(1);

    localErosionValue.textContent = "—";

  }


  function initRiver() {

    riverPoints = [];

    oxbowLakes = [];

    selectedIndex = -1;

    year = 0;

    gameOver = false;

    winStatus = null;


    const n = 32;

    const startX = 60;

    const endX = canvas.width - 60;

    const baseY = canvas.height / 2;

    const totalLengthX = endX - startX;


    for (let i = 0; i < n; i++) {

      const t = i / (n - 1);

      const x = startX + t * totalLengthX;

      const y = baseY + Math.sin(t * Math.PI * 3) * 80;

      riverPoints.push({ x, y, erosionFactor: 1.0 });

    }

    updateStatsAndDraw();

  }


  function normalizeVec(x, y) {

    const len = Math.hypot(x, y);

    if (len === 0) return { x: 0, y: 0 };

    return { x: x / len, y: y / len };

  }

  function distance(p1, p2) { return Math.hypot(p1.x - p2.x, p1.y - p2.y); }


  function computeSinuosity() {

    if (riverPoints.length < 2) return { sinuosity: 1 };

    let length = 0;

    for (let i = 1; i < riverPoints.length; i++) length += distance(riverPoints[i - 1], riverPoints[i]);

    const straight = distance(riverPoints[0], riverPoints[riverPoints.length - 1]);

    return { sinuosity: straight > 0 ? length / straight : 1 };

  }


  function pointToSegmentDistance(px, py, x1, y1, x2, y2) {

    const dx = x2 - x1, dy = y2 - y1;

    if (dx === 0 && dy === 0) return Math.hypot(px - x1, py - y1);

    const t = ((px - x1) * dx + (py - y1) * dy) / (dx * dx + dy * dy);

    const tt = Math.max(0, Math.min(1, t));

    const projX = x1 + tt * dx, projY = y1 + tt * dy;

    return Math.hypot(px - projX, py - projY);

  }


  function computeTownDistance() {

    if (riverPoints.length < 2) return Infinity;

    let minDist = Infinity;

    for (let i = 1; i < riverPoints.length; i++) {

      const p1 = riverPoints[i - 1], p2 = riverPoints[i];

      minDist = Math.min(minDist, pointToSegmentDistance(town.x, town.y, p1.x, p1.y, p2.x, p2.y));

    }

    return minDist;

  }


  function drawScene() {

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = "#f7f4e9";

    ctx.fillRect(0, 0, canvas.width, canvas.height);


    // oxbow

    ctx.lineWidth = 2;

    ctx.strokeStyle = "#7fb3d5";

    ctx.fillStyle = "rgba(127,179,213,0.5)";

    oxbowLakes.forEach(points => {

      if (points.length < 3) return;

      ctx.beginPath();

      ctx.moveTo(points[0].x, points[0].y);

      for (let i = 1; i < points.length; i++) ctx.lineTo(points[i].x, points[i].y);

      ctx.closePath();

      ctx.fill();

      ctx.stroke();

    });


    // river

    ctx.beginPath();

    ctx.moveTo(riverPoints[0].x, riverPoints[0].y);

    for (let i = 1; i < riverPoints.length; i++) ctx.lineTo(riverPoints[i].x, riverPoints[i].y);

    ctx.strokeStyle = "#4a8dd8";

    ctx.lineWidth = 4;

    ctx.stroke();


    // erosion/deposition dots

    for (let i = 1; i < riverPoints.length - 1; i++) {

      const pPrev = riverPoints[i - 1], p = riverPoints[i], pNext = riverPoints[i + 1];

      const v1 = normalizeVec(p.x - pPrev.x, p.y - pPrev.y);

      const v2 = normalizeVec(pNext.x - p.x, pNext.y - p.y);

      const cross = v1.x * v2.y - v1.y * v2.x;

      const angle = Math.acos(Math.max(-1, Math.min(1, v1.x*v2.x + v1.y*v2.y)));

      if (angle < 0.05) continue;


      const tangent = normalizeVec(pNext.x - pPrev.x, pNext.y - pPrev.y);

      const normal = { x: -tangent.y, y: tangent.x };

      const offset = 10 + 20 * angle;

      const sign = cross >= 0 ? 1 : -1;


      const outer = { x: p.x + normal.x * offset * sign, y: p.y + normal.y * offset * sign };

      const inner = { x: p.x - normal.x * offset * sign, y: p.y - normal.y * offset * sign };


      ctx.beginPath(); ctx.arc(outer.x, outer.y, 4, 0, Math.PI*2);

      ctx.fillStyle = "rgba(211,90,74,0.8)"; ctx.fill();


      ctx.beginPath(); ctx.arc(inner.x, inner.y, 4, 0, Math.PI*2);

      ctx.fillStyle = "rgba(227,201,90,0.9)"; ctx.fill();

    }


    // town

    const size = 14;

    ctx.fillStyle = "#777777";

    ctx.fillRect(town.x - size/2, town.y - size/2, size, size);

    ctx.strokeStyle = "#333"; ctx.lineWidth = 1;

    ctx.strokeRect(town.x - size/2, town.y - size/2, size, size);

    ctx.fillStyle = "#222"; ctx.font = "12px system-ui";

    ctx.fillText("城鎮", town.x + size/2 + 4, town.y + 4);


    // selected point

    if (selectedIndex >= 0) {

      const sp = riverPoints[selectedIndex];

      ctx.beginPath(); ctx.arc(sp.x, sp.y, 6, 0, Math.PI*2);

      ctx.fillStyle = "rgba(0,0,0,0.5)"; ctx.fill();

    }

  }


  function checkAndCreateOxbow() {

    const cutoff = PARAMS.cutoffDistance;

    const minGap = 4;

    let pair = null;

    for (let i = 0; i < riverPoints.length; i++) {

      for (let j = i + minGap; j < riverPoints.length; j++) {

        if (distance(riverPoints[i], riverPoints[j]) < cutoff) { pair = {i, j}; break; }

      }

      if (pair) break;

    }

    if (!pair) return;


    const {i, j} = pair;

    const ox = [];

    for (let k = i; k <= j; k++) ox.push({x: riverPoints[k].x, y: riverPoints[k].y});

    oxbowLakes.push(ox);


    const nr = [];

    for (let k = 0; k <= i; k++) nr.push({...riverPoints[k]});

    for (let k = j; k < riverPoints.length; k++) nr.push({...riverPoints[k]});

    nr[i].x = (riverPoints[i].x + riverPoints[j].x) / 2;

    nr[i].y = (riverPoints[i].y + riverPoints[j].y) / 2;

    riverPoints = nr;

  }


  function stepSimulation() {

    if (!running || gameOver) return;

    const np = riverPoints.map(p => ({...p}));


    for (let i = 1; i < riverPoints.length - 1; i++) {

      const pPrev = riverPoints[i-1], p = riverPoints[i], pNext = riverPoints[i+1];

      const v1 = normalizeVec(p.x - pPrev.x, p.y - pPrev.y);

      const v2 = normalizeVec(pNext.x - p.x, pNext.y - p.y);

      const dot = Math.max(-1, Math.min(1, v1.x*v2.x + v1.y*v2.y));

      const angle = Math.acos(dot);

      const cross = v1.x*v2.y - v1.y*v2.x;

      if (angle < 0.001) continue;


      const tangent = normalizeVec(pNext.x - pPrev.x, pNext.y - pPrev.y);

      const normal = { x: -tangent.y, y: tangent.x };

      const sign = cross >= 0 ? 1 : -1;


      const erosionMag = PARAMS.erosionBaseRate * angle * (globalFlow/3) * globalSlope * (1/globalBankResistance) * p.erosionFactor;

      const depoMag = PARAMS.depositionBaseRate * angle * (globalFlow/3) * (2 - globalSlope)/2;


      np[i].x += normal.x * erosionMag * sign;

      np[i].y += normal.y * erosionMag * sign;

      np[i].x -= normal.x * depoMag * sign * 0.5;

      np[i].y -= normal.y * depoMag * sign * 0.5;

    }


    for (let i = 1; i < np.length - 1; i++) {

      const a = np[i-1], b = np[i], c = np[i+1];

      const s = 0.2;

      np[i].x = b.x*(1-s) + ((a.x+c.x)/2)*s;

      np[i].y = b.y*(1-s) + ((a.y+c.y)/2)*s;

    }


    riverPoints = np;

    checkAndCreateOxbow();

    year += PARAMS.timeStepYears;

    updateStatsAndDraw();

    checkWinLose();

  }


  function updateStatsAndDraw() {

    const {sinuosity} = computeSinuosity();

    const td = computeTownDistance();

    yearDisplay.textContent = year + " 年";

    sinuosityDisplay.textContent = sinuosity.toFixed(2);

    targetSinuosityDisplay.textContent = PARAMS.targetSinuosity.toFixed(2);

    townDistanceDisplay.textContent = isFinite(td) ? td.toFixed(1) + " 像素" : "—";

    drawScene();

  }


  function checkWinLose() {

    const {sinuosity} = computeSinuosity();

    const td = computeTownDistance();


    if (td < 10) {

      gameOver = true; winStatus = "lose"; running = false;

      messageBox.textContent = "城鎮被曲流外側侵蝕，挑戰失敗。";

      return;

    }

    if (sinuosity >= PARAMS.targetSinuosity && year <= PARAMS.maxYears) {

      gameOver = true; winStatus = "win"; running = false;

      messageBox.textContent = "達到目標蜿蜒度且城鎮仍安全，挑戰成功。";

      return;

    }

    if (year > PARAMS.maxYears && sinuosity < PARAMS.targetSinuosity) {

      gameOver = true; winStatus = "lose"; running = false;

      messageBox.textContent = "超過年限仍未達目標蜿蜒度。";

      return;

    }

    messageBox.textContent = "持續觀察曲流發展。可點選河道調整局部侵蝕係數。";

  }


  function canvasToPoint(evt) {

    const rect = canvas.getBoundingClientRect();

    return {

      x: (evt.clientX - rect.left) * (canvas.width / rect.width),

      y: (evt.clientY - rect.top) * (canvas.height / rect.height)

    };

  }


  function handleCanvasClick(x, y) {

    let ci = -1, md = Infinity;

    for (let i = 0; i < riverPoints.length; i++) {

      const d = Math.hypot(riverPoints[i].x - x, riverPoints[i].y - y);

      if (d < md) { md = d; ci = i; }

    }

    if (md > 30) {

      selectedIndex = -1;

      localErosionSlider.disabled = true;

      localErosionValue.textContent = "—";

    } else {

      selectedIndex = ci;

      localErosionSlider.disabled = false;

      localErosionSlider.value = riverPoints[ci].erosionFactor;

      localErosionValue.textContent = riverPoints[ci].erosionFactor.toFixed(1);

    }

    drawScene();

  }


  canvas.addEventListener("click", (evt) => {

    const p = canvasToPoint(evt);

    handleCanvasClick(p.x, p.y);

  });


  flowSlider.addEventListener("input", () => {

    globalFlow = parseFloat(flowSlider.value);

    flowValue.textContent = globalFlow.toFixed(1);

  });

  slopeSlider.addEventListener("input", () => {

    globalSlope = parseFloat(slopeSlider.value);

    slopeValue.textContent = globalSlope.toFixed(1);

  });

  bankResSlider.addEventListener("input", () => {

    globalBankResistance = parseFloat(bankResSlider.value);

    bankResValue.textContent = globalBankResistance.toFixed(1);

  });

  localErosionSlider.addEventListener("input", () => {

    if (selectedIndex < 0) return;

    const v = parseFloat(localErosionSlider.value);

    riverPoints[selectedIndex].erosionFactor = v;

    localErosionValue.textContent = v.toFixed(1);

  });


  startBtn.addEventListener("click", () => {

    if (gameOver) initRiver();

    running = true;

    if (!timerId) timerId = setInterval(stepSimulation, 80);

  });

  pauseBtn.addEventListener("click", () => { running = false; });

  resetBtn.addEventListener("click", () => {

    running = false; gameOver = false; winStatus = null; year = 0;

    initRiver();

    messageBox.textContent = "已重新開始，可調整參數後再按「開始」。";

  });


  initSliders();

  initRiver();
